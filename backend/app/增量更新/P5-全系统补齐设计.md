## ✅ Phase 0: SceneBus 事件标签注入

### 问题
Phase 3/4/5 依赖 SceneBus entries 上的语义 tags（`COMBAT_END`、`CRISIS`、`REST`）来检测事件。
但当前引擎执行后**不向 SceneBus 写入带 tags 的 entry**——只有 `narrative_hints` 进 L7。

### 改动文件
`app/game_core/orchestration/tick_coordinator.py` — `_record_action()` 之后

### 设计

在 `TickCoordinator.process()` 中，引擎执行完毕、`_record_action()` 之后，
根据 `PipelineResult.action_type` 和 `metadata` 向 SceneBus 写入 ENGINE entry：

```python
def _emit_action_tags(self, result: PipelineResult) -> None:
    """Write semantic tags to SceneBus for downstream hook consumption."""
    tag_map: dict[str, list[str]] = {
        "rest_long": ["REST", "LONG_REST"],
        "rest_short": ["REST", "SHORT_REST"],
        "start_combat": ["COMBAT"],
        "end_combat": ["COMBAT_END"],
        "navigate": ["NAVIGATION"],
        "skill_check": ["SKILL_CHECK"],
    }
    tags = tag_map.get(result.action_type, [])
    if not tags:
        return

    # 从 metadata 推断额外 tags
    if result.metadata.get("target_defeated"):
        tags.append("COMBAT_END")
    if not result.success:
        tags.append("FAILURE")

    self.scene_bus.add_entry({
        "source": "ENGINE",
        "content": f"[{result.action_type}] {'; '.join(result.narrative_hints[:2]) if result.narrative_hints else ''}",
        "visibility": "system",
        "tags": tags,
    })
```

调用时机：`process()` 中 `_record_action(result)` 之后立即调用。

**注意**：`end_combat` 可能不是一个独立的 action_type。需要根据战斗系统的实际实现确认。
如果战斗结束是通过 metadata 中的 `target_defeated` 等字段判断，则在 tag_map 之外补充检测。

**预计改动**：~30 行。

### 2.4 私聊临时场景生成

PrivateChatCoordinator Step 1 中，使用 DynamicSubAreaManager 创建临时私密子地点：

```python
# private_chat.py — Step 1 新增
import random

_PRIVATE_CHAT_SCENES: dict[str, list[dict[str, str]]] = {
    "tavern": [
        {"name": "酒馆二楼的小包间", "description": "一间昏暗的小房间，只有一盏油灯和两把椅子。楼下的喧闹声被厚重的木门隔绝在外。"},
        {"name": "后门外的僻静角落", "description": "酒馆后门外的小巷，头顶的木质屋檐遮住了月光。远处传来微弱的酒馆歌声。"},
    ],
    "town": [
        {"name": "城墙上的角落", "description": "城墙的一处凹角，可以俯瞰整个小镇的灯火。风很大，但没有人会来这里。"},
    ],
    "forest": [
        {"name": "远离营地的老树下", "description": "一棵巨大的橡树，粗壮的根部形成了天然的座椅。篝火的光在这里只剩下微弱的橙色。"},
    ],
    "default": [
        {"name": "僻静处", "description": "一个远离人群的安静角落。"},
    ],
}

def _create_private_scene(self, npc_id: str, area_id: str) -> dict[str, Any] | None:
    """创建私聊临时子地点。"""
    if not self._state.has_slice("areas"):
        return None

    # 根据 area tags 匹配场景模板
    area_template = self._world.maps.get(area_id) if self._world.has_registry("maps") else None
    area_tags = getattr(area_template, "tags", []) if area_template else []

    scene_list = _PRIVATE_CHAT_SCENES.get("default", [])
    for tag in area_tags:
        tag_lower = str(tag).lower()
        if tag_lower in _PRIVATE_CHAT_SCENES:
            scene_list = _PRIVATE_CHAT_SCENES[tag_lower]
            break

    template = random.choice(scene_list)
    tick = self._state.time.absolute_tick() if self._state.has_slice("time") else 0
    sub_area_id = f"_private_{npc_id}_{tick}"

    sub_area_data = {
        "id": sub_area_id,
        "name": template["name"],
        "description": template["description"],
        "tags": ["PRIVATE_CHAT", "TEMPORARY"],
        "type": "private",
        "resident_npcs": [npc_id],
        "source": "private_chat",
        "expiry": -1,  # 手动 expire
    }
    self._state.areas.add_temporary_sub_area(area_id, sub_area_data)
    return sub_area_data
```

## ✅ Phase 3: SharedExperience 录入 Hook

### 前置依赖

### 新文件
`app/game_core/orchestration/hooks/shared_experience.py`

### 设计

```python
class SharedExperienceHook(NoOpSettlementHook):
    """记录队伍共同经历。P55，在 RelationshipHook(P65) 之前执行。

    数据源：
    - SceneBus entries 的 tags（Phase 0 写入的 ENGINE tags）
    - action_log 中的 action_type + success
    """

    HOOK_PRIORITY = 55
    HOOK_NAME = "shared_experience"

    async def execute(self, context: SettlementContext) -> HookResult:
        if not context.state.has_slice("party"):
            return HookResult()
        members = context.state.party.members
        if not isinstance(members, dict) or not members:
            return HookResult()

        experience = self._detect_experience(context)
        if experience is None:
            return HookResult()

        context.state.party.record_experience(experience)
        context.record_change(StateChange(
            slice="party", operation="append",
            path="shared_experiences", value=experience,
        ))
        return HookResult()

    def _detect_experience(self, context: SettlementContext) -> dict | None:
        # 从 action_log 检测（比 SceneBus tags 更可靠）
        action_types = {a.get("type") for a in context.action_log}

        day = context.state.time.snapshot().get("day", 1) if context.state.has_slice("time") else 1
        location = context.state.player.current_area if context.state.has_slice("player") else ""

        # 也检查 SceneBus tags

        for entry in context.scene_bus.get_entries():
            bus_tags.update(getattr(entry, "tags", []))

        # 战斗
        if "COMBAT_END" in bus_tags or "end_combat" in action_types:
            return {
                "type": "combat",
                "summary": f"Day {day} combat at {location}",
                "day": day, "location": location,
                "participants": participants,
                "critical_moment": any(a.get("type") == "end_combat" and "boss" in str(a) for a in context.action_log),
                "emotion_tags": ["danger", "relief"],
            }

        # 任务相关
        if "advance_quest" in action_types:
            return {
                "type": "discovery",
                "summary": f"Day {day} quest progress at {location}",
                "day": day, "location": location,
                "participants": participants,
                "critical_moment": False,
                "emotion_tags": ["achievement"],
            }

        # 长休（轻量经历，非 critical）
        if "rest_long" in action_types:
            return {
                "type": "rest",
                "summary": f"Day {day} camp at {location}",
                "day": day, "location": location,
                "participants": participants,
                "critical_moment": False,
                "emotion_tags": ["rest"],
```

```python
class CampfireHook(NoOpSettlementHook):
    """长休后营火对话。P72，在 TimeAdvanceHook(P70) 之后执行。"""

    HOOK_PRIORITY = 72
    HOOK_NAME = "campfire"

    async def execute(self, context: SettlementContext) -> HookResult:
        # 1. 检测长休（从 action_log 中）
        has_rest = any(a.get("type") == "rest_long" for a in context.action_log)
        if not has_rest:
            return HookResult()

        # 2. 检查队伍
        if not context.state.has_slice("party"):
            return HookResult()
        members = context.state.party.members
        if not isinstance(members, dict) or not members:
            return HookResult()

        # 3. 触发概率
        day = context.state.time.snapshot().get("day", 1) if context.state.has_slice("time") else 1
        all_exp = context.state.party.get_shared_experiences()
        today_exp = [e for e in all_exp if isinstance(e, dict) and e.get("day") == day]
        has_major = any(e.get("critical_moment") for e in today_exp)

        if not has_major and random.random() > 0.3:
            return HookResult()

        # 4. 选择经历
        selected = self._select_experience(all_exp, today_exp, day)
        if selected is None:
            return HookResult()

        # 5. SSE 事件
        events = [SSEEvent(
            event_type="campfire_start",
            payload={
                "experience": selected,
                "participants": list(members.keys()),
            },
        )]

        # 6. 关系巩固（approval +3）
        if context.state.has_slice("relations"):
            for member_id in members:
                disp = context.state.relations.npc_dispositions
                if isinstance(disp, dict) and member_id in disp:
                    current = disp[member_id].get("approval", 0) if isinstance(disp[member_id], dict) else 0
                    new_val = min(100, current + 3)
                    context.state.relations.set_disposition_value(member_id, "approval", new_val)
                    context.record_change(StateChange(
                        slice="relations", operation="set",
                        path=f"npc_dispositions.{member_id}.approval", value=new_val,
                    ))

        return HookResult(events=events)

    def _select_experience(self, all_exp, today_exp, day) -> dict | None:
        # 优先级 1：今天的重大经历
        critical = [e for e in today_exp if e.get("critical_moment")]
        if critical: return critical[-1]
        # 优先级 2：今天的任何经历
        if today_exp: return today_exp[-1]
        # 优先级 3：最近 7 天
        recent = [e for e in all_exp if isinstance(e, dict) and day - e.get("day", 0) <= 7]
        if recent:
            recent.sort(key=lambda e: (e.get("critical_moment", False), len(e.get("emotion_tags", []))), reverse=True)
            return recent[0]
        # 优先级 4：随机旧经历
        if all_exp: return random.choice(all_exp)
        return None
```

```python
@dataclass(slots=True)
class RecruitResult:
    success: bool
    reason: str = ""  # "recruited" / "party_full" / "npc_refuses" / "already_member" / ...

class CompanionManager:
    MAX_PARTY_SIZE = 4

    def __init__(self, world: WorldInstance, state: StateContainer) -> None:
        self._world = world
        self._state = state

    def recruit(self, npc_id: str) -> RecruitResult:
        """招募 NPC 为队友。
        前置条件：
        - NPC 存在 + 有 RECRUITABLE tag
        - 关系阶段 >= acquaintance
        - approval > 0
        - 队伍未满
        """
        if not self._world.has_registry("characters"):
            return RecruitResult(False, "no_character_registry")
        profile = self._world.characters.get(npc_id)
        if profile is None:
            return RecruitResult(False, "npc_not_found")

        tags = getattr(profile, "tags", [])
        if "recruitable" not in [t.lower() for t in tags]:
            return RecruitResult(False, "not_recruitable")

        # 关系检查
        if self._state.has_slice("relations"):
            stage = self._state.relations.get_stage(npc_id)
            if stage == "stranger":
                return RecruitResult(False, "stranger")
            disp = self._state.relations.npc_dispositions.get(npc_id, {})
            if isinstance(disp, dict) and disp.get("approval", 0) <= 0:
                return RecruitResult(False, "npc_refuses")

        # 队伍容量
        if self._state.has_slice("party"):
            members = self._state.party.members
            if npc_id in members:
                return RecruitResult(False, "already_member")
            if len(members) >= self.MAX_PARTY_SIZE:
                return RecruitResult(False, "party_full")
            self._state.party.add_member(npc_id, {
                "name": getattr(profile, "name", npc_id),
                "class_id": getattr(profile, "class_id", ""),
                "recruited_tick": self._state.time.absolute_tick() if self._state.has_slice("time") else 0,
            })
        return RecruitResult(True, "recruited")

    def dismiss(self, npc_id: str) -> RecruitResult:
        """让队友离队。"""
        if not self._state.has_slice("party"):
            return RecruitResult(False, "no_party")
        if npc_id not in self._state.party.members:
            return RecruitResult(False, "not_member")

        self._state.party.remove_member(npc_id)

        # 恢复日程位置（按当前时段）
        # NpcScheduleHook 会在下次结算时自动处理

        return RecruitResult(True, "dismissed")

    def force_leave(self, npc_id: str, reason: str) -> RecruitResult:
        """队友主动离队（关系恶化）。
        由 RelationshipHook 在检测到负面阶段时调用。
        """
        result = self.dismiss(npc_id)
        if result.success:
            result.reason = f"force_leave:{reason}"
        return result
```

```python
def _condition_met(self, state, condition) -> tuple[bool, int]:
    # ... existing types ...

        return self._check_npc_talked(state, params), 0
    if condition_type == "item_obtained":

@staticmethod
def _check_npc_talked(state: StateContainer, params: Mapping[str, Any]) -> bool:
    """检查是否和指定 NPC 交谈过。通过 FlagSlice 的 talked_to_{npc_id} flag 判定。"""
    npc_id = _coerce_non_empty_string(params.get("npc_id"))
    if npc_id is None or not state.has_slice("flags"):
        return False
    return bool(state.flags.get(f"talked_to_{npc_id}"))

@staticmethod
def _check_item_obtained(state: StateContainer, params: Mapping[str, Any]) -> bool:
    """检查玩家是否持有指定物品。"""
    item_id = _coerce_non_empty_string(params.get("item_id"))
    if item_id is None or not state.has_slice("player"):
        return False
    inventory = state.player.snapshot().get("inventory", [])
    return any(
        isinstance(item, dict) and item.get("item_id") == item_id
        for item in inventory
    )

@staticmethod
def _check_kill_count(state: StateContainer, params: Mapping[str, Any]) -> bool:
    """检查击杀计数是否达标。通过 FlagSlice 的 kill_count_{monster_type} 判定。"""
    monster_type = _coerce_non_empty_string(params.get("monster_type"))
    required = int(params.get("count", 1))
    if monster_type is None or not state.has_slice("flags"):
        return False
    current = state.flags.get(f"kill_count_{monster_type}", 0)
    try:
        return int(current) >= required
    except (TypeError, ValueError):
        return False
```

- ⚠️ `kill_count_{monster_type}`：评估器已就绪，**写入侧（CombatHandler）未做**，当前需通过 set_flag 命令手动触发
